{{- /* Highlighted code block from a file in the post's page bundle, shown as a card with a header bar (title, language, download link) above the code: {{< code-include source="script.sh" >}}. source file beside the post's index.md (required) language Chroma lexer name; defaults to the file extension (e.g. "sh", "py") options Hugo highlight options (e.g. "linenos=table,hl_lines=3-5") collapse "true" to show only the header until the reader expands it, "open" to make it collapsible but start expanded title header text; defaults to the source file name Styled in assets/css/custom/30-post-blocks.css. The inner .highlight-wrapper picks up Blowfish's copy button. */ -}} {{- $source := .Get "source" -}} {{- if not $source -}} {{- errorf "%s: code-include needs a source file" .Position -}} {{- else -}} {{- with .Page.Resources.Get $source -}} {{- $language := $.Get "language" | default (strings.TrimPrefix "." (path.Ext $source)) -}} {{- $collapse := $.Get "collapse" | default "false" | string -}} {{- if not (in (slice "true" "open" "false") $collapse) -}} {{- errorf "%s: code-include collapse must be \"true\", \"open\", or \"false\", not %q" $.Position $collapse -}} {{- end -}} {{- $collapsible := ne $collapse "false" -}} {{- $header := printf `%s%sDownload` (htmlEscape ($.Get "title" | default $source)) (htmlEscape $language) .RelPermalink (htmlEscape $source) -}} {{- $code := printf `
%s
` (highlight (trim .Content "\n\r") $language ($.Get "options")) -}} {{- if $collapsible -}}
{{ $header | safeHTML }} {{- $code | safeHTML -}}
{{- else -}}
{{ $header | safeHTML }}
{{- $code | safeHTML -}}
{{- end -}} {{- else -}} {{- errorf "%s: code-include file %q not found beside the post" $.Position $source -}} {{- end -}} {{- end -}}
