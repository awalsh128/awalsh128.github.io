{{- /* Renders LaTeX between $...$ (inline) or $$...$$ (block) to KaTeX HTML at build time. */ -}} {{- /* siunitx-style unit macros, since KaTeX doesn't ship siunitx: \unit{ms}, \qty{43.2}{min}, \num{43200}. \si and \SI are aliases for the older siunitx names. */ -}} {{- $macros := dict "\\unit" "\\mathrm{#1}" "\\qty" "#1\\,\\mathrm{#2}" "\\num" "#1" "\\si" "\\unit{#1}" "\\SI" "\\qty{#1}{#2}" "\\percent" "\\%" }} {{- $opts := dict "output" "htmlAndMathml" "displayMode" (eq .Type "block") "macros" $macros }} {{- with try (transform.ToMath .Inner $opts) }} {{- with .Err }} {{- errorf "Unable to render math %q in %s: %s" $.Inner $.Page.File.Path . }} {{- else }} {{- .Value }} {{- $.Page.Store.Set "hasMath" true }} {{- end }} {{- end -}}
